\section{Contexte et Enjeux}

% --- SLIDE 3 : CONTEXTE ---
\begin{frame}{Contexte du Projet}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Le Projet CIGEO :}
        \begin{itemize}
            \item Stockage géologique à \textbf{500m de profondeur} (Bure).
            \item Infrastructure : Plus de \textbf{270 km de galeries}.
        \end{itemize}
        
        \vspace{0.5cm}
        
        \textbf{La Problématique :}
        \begin{itemize}
            \item Apparition de fissures (cisaillement géologique).
            \item Inspection manuelle répétitive et chronophage.
        \end{itemize}
        
        \column{0.5\textwidth}
        \begin{figure}
            \centering
            \includegraphics[width=0.9\textwidth]{cigeo.jpg}
            \caption{Vue du laboratoire à Bure}
        \end{figure}
    \end{columns}
\end{frame}

% --- SLIDE 3 BIS : ENJEUX ---
\begin{frame}{Enjeux et Défis Techniques}
    \begin{block}{Enjeux Opérationnels}
        \begin{itemize}
            \item \textbf{Industrialisation :} Passer d'une inspection manuelle ponctuelle à une surveillance robotisée systématique et reproductible.
            \item \textbf{Volume à traiter :} Couvrir efficacement les 270 km de galeries prévus pour le projet CIGEO.
        \end{itemize}
    \end{block}

    \vspace{0.5cm}

    \begin{block}{Défis Scientifiques}
        \begin{itemize}
            \item \textbf{Environnement :} Absence totale de GPS à 500m de profondeur $\rightarrow$ Nécessité d'une localisation relative robuste (SLAM/AMCL).
            \item \textbf{Milieu Complexe : }Conditions difficiles (poussière, obstacles).
        \end{itemize}
    \end{block}
\end{frame}

% --- SLIDE 5 : NAVIGATION ---
\begin{frame}{État de l'Art : Navigation \& Positionnement}
    \textbf{Architecture Logicielle :}
    \begin{itemize}
        \item Migration complète vers \textbf{ROS2 (Humble)}.
        \item Abandon des API propriétaires $\rightarrow$ Modularité accrue.
    \end{itemize}

    \vspace{0.5cm}

    \textbf{Stratégie de Localisation :}
    \begin{itemize}
        \item \textbf{Problème :} Le SLAM (cartographie simultanée) dérive trop en tunnel ("Drift").
        \item \textbf{Solution retenue : AMCL} (Adaptive Monte Carlo Localization).
        \item \textit{Principe :} Se recaler sur une carte statique connue via le LiDAR 2D.
    \end{itemize}
\end{frame}

\section{Pôle Robotique (Vincent \& Paul-Antoine)}

% --- SLIDE : TRAVAUX S1 ROBOTIQUE ---
\begin{frame}{Robotique S1 : Reconstruction de l'Architecture}
    \begin{block}{Incident Critique}
        Perte totale de l'image système de l'année précédente sans sauvegarde.
    \end{block}

    \vspace{0.3cm}

    \textbf{Actions correctives réalisées :}
    \begin{itemize}
        \item Reconstruction intégrale de l'environnement ROS2 (Workspaces, dépendances).
        \item \textbf{Interface CAN :} Patch du driver \texttt{scout\_base} pour reconnaissance matérielle spécifique.
        \item \textbf{Caméra PTZ :} Rétablissement IP et création de scripts d'automatisation.
    \end{itemize}

    \vspace{0.3cm}
    \textbf{État des capteurs :}
    \begin{itemize}
        \item[\textcolor{green}{\checkmark}] Caméra ZED2, PTZ, Odométrie Base.
        \item[\textcolor{red}{\textbf{!}}] \textbf{LiDAR :} Erreurs d'initialisation (Panne matérielle suspectée).
    \end{itemize}
\end{frame}

% --- SLIDE : OBJECTIFS S2 ROBOTIQUE ---
\begin{frame}{Robotique : Objectifs Semestre 2}
    L'objectif est de passer d'un robot "télécommandé" à un robot "supervisé".

    \begin{enumerate}
        \item \textbf{Fiabilisation de la Navigation :}
        \begin{itemize}
            \item Résoudre le problème matériel du LiDAR.
            \item Valider la chaîne de transformation (TF) complète.
        \end{itemize}
        
        \item \textbf{Implémentation AMCL :}
        \begin{itemize}
            \item Réussir la localisation sur une carte pré-enregistrée (TechLab puis Bure).
            \item Minimiser la dérive (Drift) sur les longs trajets linéaires.
        \end{itemize}

        \item \textbf{Validation sur site (Février) :}
        \begin{itemize}
            \item Assurer un cycle \textit{Avancer $\rightarrow$ Stop $\rightarrow$ Photo} robuste pour la collecte de données.
        \end{itemize}
    \end{enumerate}
\end{frame}

% --- SLIDE 10 : LIMITATIONS ---
\begin{frame}{Contraintes et Limitations Actuelles}
    \begin{itemize}
        \setlength\itemsep{1em}
        \item[\textbf{--}] \textbf{Matériel 360° :} Équipement prêté par l'ANDRA intégration au robot impossible.
        
        \item[\textbf{--}] \textbf{Communication :} Système de hot spot testé au techlab mais non définitif.
        
        \item[\textbf{--}] \textbf{Maturité :} Système global en retrait par rapport à la fin du projet 2024-2025.

        \item[\textbf{--}] \textbf{Localisation :} La chaîne complète (création de carte + navigation) reste à maîtriser.

        \item[\textbf{--}] \textbf{LIDAR :} Problème de matériel qui nous empêche de créer des cartes.
    \end{itemize}
\end{frame}
