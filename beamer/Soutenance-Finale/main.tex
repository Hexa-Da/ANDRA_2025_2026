\section{Contexte et Architecture}
% ============================================================

\begin{frame}{Problématique : Inspection du projet Cigéo}
    \begin{columns}
        \column{0.6\textwidth}
        \textbf{Le besoin de l'ANDRA :}
        \begin{itemize}
            \item Surveiller \textbf{270 km} de galeries à 500\,m de profondeur
            \item Détecter les fissures dues au cisaillement géologique
            \item Remplacer une inspection manuelle chronophage
        \end{itemize}
        \vspace{0.5cm}
        \begin{center}
            \begin{minipage}{0.9\linewidth}
                \begin{block}{Objectif du projet}
                    Un robot autonome capable de naviguer en galerie, détecter les fissures par IA et \textbf{géolocaliser chaque détection} sur la carte.
                \end{block}
            \end{minipage}
        \end{center}
        \column{0.4\textwidth}
        \vspace{0.5cm}
        \begin{figure}
            \centering
            \includegraphics[width=\textwidth]{cigeo.jpg}
            \caption{Laboratoire souterrain de Bure}
        \end{figure}
    \end{columns}
\end{frame}

\begin{frame}{Architecture Matérielle}
    \begin{columns}
        \column{0.52\textwidth}
        \begin{itemize}
            \item \textbf{Base :} Agilex Scout Mini (odométrie, CAN)
            \item \textbf{Calcul :} Jetson Orin Nano (GPU)
            \item \textbf{Vision :}
            \begin{itemize}
                \item Caméra PTZ Marshall CV-605
                \item ZED2i (IMU)
                \item Caméra 360° Ricoh Theta (prêt ANDRA)
            \end{itemize}
            \item \textbf{Positionnement :} YDLidar TG15
            \item \textbf{Logiciel :} ROS2 Humble + Nav2
        \end{itemize}
        \column{0.48\textwidth}
        \begin{figure}
            \centering
            \includegraphics[width=0.75\textwidth]{images/avant.JPG}
            \caption{Robot dans sa configuration finale}
        \end{figure}
    \end{columns}
\end{frame}

\begin{frame}{État de l'art — Détection de fissures}
    \begin{columns}
        \column{0.55\textwidth}
        \textbf{Problème des bases de données génériques :}
        \begin{itemize}
            \item Faux positifs (câbles, tuyaux et interstices).
            \item Modèles open data inefficaces en production sur le site ANDRA.
        \end{itemize}
        \textbf{Solution 2024-2025 (reprise) :}
        \begin{itemize}
            \item[$\checkmark$] Dataset spécifique ANDRA : $\approx$300 images annotées manuellement issues de la PTZ.
            \item[$\checkmark$] Modèle \textbf{YOLOv11} entraîné sur ce dataset.
            \item[$\checkmark$] Faux positifs (câbles, barrières) éliminés.
        \end{itemize}
        \begin{block}{Limite actuelle}
            Modèle valide uniquement sur mur de type \textbf{GER}.
        \end{block}
        \column{0.35\textwidth}
        \begin{figure}
            \centering
            \includegraphics[width=0.6\textwidth]{images/example_mauvaise_detection_fissure_2.png}
            \caption{Détection OpenData}
        \end{figure}
    \end{columns}
\end{frame}

\section{Semestre 2 : Résolutions Techniques}
% ============================================================

\begin{frame}{S2 — Résolution du problème LiDAR}
    \begin{columns}
        \column{0.6\textwidth}
        \begin{block}{Correction matérielle}
            Branchement de secours via \textbf{USB-C}.\\
            Port série : \texttt{/dev/ttyTHS1} → \texttt{/dev/ttyUSB0}.
        \end{block}
        \begin{block}{Correction logicielle}
            Mauvais fichier de configuration \texttt{G4.yaml} → adoption du fichier officiel \textbf{\texttt{TG.yaml}} (spécifique au TG15).
        \end{block}
        \vspace{0.3cm}
        \textbf{Résultat :} Scan 360° fonctionnel dès le 18 février 2026. Navigation AMCL et cartographie SLAM désormais possibles.
        \column{0.4\textwidth}
        \begin{figure}
            \centering
            \includegraphics[width=0.85\textwidth]{images/chipset_lidar_repare.jpg}
            \caption{Carte Radar\_Con réparée}
        \end{figure}
    \end{columns}
\end{frame}

\begin{frame}{S2 — Optimisation IA : TensorRT \& Docker}
    Pour avoir une détection en temps réel sur la Jetson Orin Nano :
    \begin{itemize}
        \item \textbf{Conteneurisation :} image Docker isolant CUDA, ROS2 et Ultralytics.
        \item \textbf{TensorRT :} conversion \texttt{best.pt} $\rightarrow$ \texttt{best.engine}.
        \item \textbf{Mode segmentation :} passage de bounding boxes à des contours de fissures.
    \end{itemize}
    \vspace{0.5cm}
    \begin{table}
        \centering
        \begin{tabular}{lccc}
            \toprule
            \textbf{Format} & \textbf{Temps / image} & \textbf{Images/s} & \textbf{Gain} \\
            \midrule
            PyTorch (\texttt{.pt}) & 125--200\,ms & 5--8 & $1\times$ \\
            \textbf{TensorRT (\texttt{.engine})} & \textbf{33--40\,ms} & \textbf{25--30} & \textbf{$\approx\times4$} \\
            \bottomrule
        \end{tabular}
        \caption{Performances d'inférence sur Jetson Orin Nano (chaîne complète)}
    \end{table}
\end{frame}

\begin{frame}{S2 — Reconstruction de l'arbre TF et de l'EKF}
    \textbf{Symptôme :} trajectoire non planaire dans RViz (visible dès la 1\textsuperscript{ère} descente).

    \vspace{0.3cm}
    \textbf{Corrections apportées :}
    \begin{itemize}
        \item Reprise complète des frames et transformations statiques (distances inter-capteurs).
        \item Suppression de la frame \texttt{world} (redondante avec \texttt{odom}).
        \item EKF simplifié : position/vitesse depuis \texttt{/odom\_robot} uniquement ; IMU ZED pour \texttt{vyaw}.
        \item SLAM Toolbox paramétré avec des frames strictes : \texttt{map} → \texttt{odom} → \texttt{base\_link}.
    \end{itemize}
    \vspace{0.2cm}
    \begin{block}{Résultat}
        Trajectoire cohérente sur plan horizontale. La navigation AMCL et la cartographie SLAM deviennent donc exploitables. Pilotage du robot depuis RViz possible.
    \end{block}
\end{frame}

\section{Semestre 2 : Navigation Autonome et Patrouille}
% ============================================================

\begin{frame}{S2 — Optimisation Mécanique : Platines 3D}
    \begin{columns}
        \column{0.52\textwidth}
        \textbf{Problème :} les piliers de la tourelle masquaient le LiDAR, créant des angles morts qui empêchaient une cartographie et une localisation fiable.

        \vspace{0.3cm}
        \textbf{Solutions testées :}
        \begin{itemize}
            \item Prototype plexiglas (abandonné).
            \item \textbf{Platine LiDAR :} capteur hors des piliers.
            \item \textbf{Platine PTZ :} caméra à l'arrière.
        \end{itemize}
        \vspace{0.2cm}
        \textbf{Résultat :} scans dégagés, suppression du paramètre \texttt{laser\_mount\_yaw}, cartes exploitables en laboratoire et localisation performante.
        \column{0.48\textwidth}   
        \vspace{-0.2cm}
        \begin{figure}
            \centering
            \includegraphics[width=0.5\textwidth]{images/assemblage.png}
            \vspace{-0.2cm}
            \caption{Prototype}
        \end{figure}
        \vspace{-0.2cm}
        \begin{figure}
            \centering
            \includegraphics[width=0.5\textwidth]{images/montage_final.jpg}
            \caption{Tourelle originale}
        \end{figure}
    \end{columns}
\end{frame}

\begin{frame}{S2 — Navigation Autonome par Waypoints (AMCL + Nav2)}
    \begin{columns}
        \column{0.45\textwidth}
        \begin{itemize}
            \item \textbf{Localisation :} AMCL recalé sur carte statique via EKF.
            \item \textbf{Interface :} RViz (\texttt{2D Nav Goal}).
            \item \textbf{Mission :} Script de mission pour l'envoi séquentiel des waypoints.
            \item \textbf{Validation :} Séries de 10 waypoints réussies au TechLab.
        \end{itemize}
        \vspace{0.2cm}
        \begin{block}{Évitement d'obstacles}
            Esquive native Nav2 privilégiée (l'arrêt d'urgence générait trop de faux positifs).
        \end{block}
        \column{0.55\textwidth}
        \vspace{0.5cm}
        \begin{figure}
            \centering
            \includegraphics[width=\textwidth]{screen_rviz.png}
            \caption{Interface RViz2}
        \end{figure}
    \end{columns}
\end{frame}

\begin{frame}{S2 — Fusion Navigation + Acquisition : \texttt{patrouille\_autonome}}
    \begin{block}{Principe : exécution en parallèle}
        \texttt{fusion\_photo/video\_navigation} lance \texttt{trajectoire\_mission} et \texttt{sequence\_photo/video}.
    \end{block}
    \vspace{0.25cm}
    \begin{columns}[t]
        \column{0.5\textwidth}
        \textbf{Mode Photo-Navigation :}
        \begin{enumerate}
            \item Nav2 suit la trajectoire YAML en continu.
            \item \texttt{sequence\_photo} : 5 captures PTZ déclenchées tous les 1\,m.
            \item \texttt{image\_subscriber} : analyse YOLO en parallèle sur \texttt{/photo\_topic}.
        \end{enumerate}
        \column{0.5\textwidth}
        \textbf{Mode Vidéo-Navigation :}
        \begin{enumerate}
            \item Nav2 suit la trajectoire YAML en continu.
            \item Enregistrement vidéo continu, balayage PTZ sinusoïdal.
            \item Analyse différée : \texttt{video\_publisher} découpe la vidéo une fois terminée.
        \end{enumerate}
    \end{columns}
    \vspace{0.5cm}
    \centering
    \textit{Fin de mission : arrêt des processus, commande STOP, retour PTZ Home.}
\end{frame}

\begin{frame}{S2 — Datasets et modèle 360°}
    \begin{columns}[t]

        \begin{itemize}

\begin{frame}{S2 — Géolocalisation des Fissures sur Carte}
    \begin{columns}
        \column{0.58\textwidth}

        \vspace{0.2cm}

        \texttt{image\_subscriber} détecte une fissure\\
        $\downarrow$ publie sur \texttt{/position\_detectee}\\
        $\downarrow$ \texttt{report\_fissures} convertit \texttt{odom} $\rightarrow$ \texttt{map} (TF)\\

\section{Conclusion et Perspectives}
% ============================================================

\begin{frame}{Conclusion}
    \begin{block}{Bilan 2025-2026}
        Transition réussie d'une infrastructure perdue vers une plateforme de patrouille autonome.
    \end{block}
    \vspace{0.4cm}
    \textbf{Trois étapes clés franchies :}
    \begin{enumerate}
        \item \textbf{Acquisition fiable :} séquence photos ou vidéo, couverture de l'arche validée en tunnel.
        \item \textbf{Navigation autonome :} AMCL + Nav2 sur carte, fusion nav/acquisition prototypée.
        \item \textbf{Pipeline complet :} détection PTZ TensorRT ($\times$4) $\rightarrow$ géolocalisation $\rightarrow$ PNG horodaté.
    \end{enumerate}
    \vspace{0.4cm}
    \centering
    \textbf{Chiffres clés :} 11 nœuds $\cdot$ 12 guides $\cdot$ 451 PTZ $\cdot$ 344 img. 360° $\cdot$ $\times$4 TensorRT.
\end{frame}

    \begin{columns}
